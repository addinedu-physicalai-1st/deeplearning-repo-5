│ - ConnectionManager 클래스로 다수 클라이언트 관리                            │
│ - 메시지 타입: log (5초 주기), action_guide (GenAI 결과), system (시스템     │

│ 3. 데이터베이스 스키마 (MySQL)                                               │
│                                                                              │
│ CREATE DATABASE pet_care_v3 CHARACTER SET utf8mb4 COLLATE                    │
│ utf8mb4_unicode_ci;                                                          │
│                                                                              │
│ -- 회원                                                                      │
│ CREATE TABLE members (                                                       │
│     id INT AUTO_INCREMENT PRIMARY KEY,                                       │
│     name VARCHAR(50) NOT NULL,                                               │
│     email VARCHAR(100),                                                      │
│     phone VARCHAR(20),                                                       │
│     created_at DATETIME DEFAULT CURRENT_TIMESTAMP                            │
│ );                                                                           │
│                                                                              │
│ -- 반려동물                                                                  │
│ CREATE TABLE pets (                                                          │
│     id INT AUTO_INCREMENT PRIMARY KEY,                                       │
│     member_id INT NOT NULL,                                                  │
│     name VARCHAR(50) NOT NULL,                                               │
│     species ENUM('dog', 'cat') NOT NULL,                                     │
│     breed VARCHAR(100),                                                      │
│     age INT,                                                                 │
│     created_at DATETIME DEFAULT CURRENT_TIMESTAMP,                           │
│     FOREIGN KEY (member_id) REFERENCES members(id) ON DELETE CASCADE         │
│ );                                                                           │
│                                                                              │
│ -- 감정 로그 (5초 주기)                                                      │
│ CREATE TABLE emotion_logs (                                                  │
│     id BIGINT AUTO_INCREMENT PRIMARY KEY,                                    │
│     pet_id INT,                                                              │
│     timestamp DATETIME NOT NULL,                                             │
│     object_type ENUM('none', 'dog', 'cat') NOT NULL DEFAULT 'none',          │
│     emotion ENUM('none', 'positive', 'negative') NOT NULL DEFAULT 'none',    │
│     emotion_detail VARCHAR(50),          -- "공포/공격성", "불안/슬픔",      │
│ "화남/불쾌"                                                                  │
│     emotion_conf FLOAT DEFAULT 0.0,                                          │
│     fps INT DEFAULT 0,                                                       │
│     order_text TEXT DEFAULT 'none',      -- GenAI 행동지시, 없으면 'none'    │
│     raw_json JSON,                                                           │
│     created_at DATETIME DEFAULT CURRENT_TIMESTAMP,                           │
│     FOREIGN KEY (pet_id) REFERENCES pets(id) ON DELETE SET NULL,             │
│     INDEX idx_timestamp (timestamp),                                         │
│     INDEX idx_pet_emotion (pet_id, emotion)                                  │
│ );                                                                           │
│                                                                              │
│ -- GenAI 행동지시 이력                                                       │
│ CREATE TABLE action_guides (                                                 │
│     id BIGINT AUTO_INCREMENT PRIMARY KEY,                                    │
│     log_id BIGINT,                                                           │
│     emotion_trigger VARCHAR(50) NOT NULL,                                    │
│     vlm_description TEXT,                                                    │
│     action_text TEXT NOT NULL,                                               │
│     control_cmd VARCHAR(50),                                                 │
│     control_param VARCHAR(50),                                               │
│     created_at DATETIME DEFAULT CURRENT_TIMESTAMP,                           │
│     FOREIGN KEY (log_id) REFERENCES emotion_logs(id) ON DELETE SET NULL      │
│ );                                                                           │
│                                                                              │
│ 시드 데이터:                                                                 │
│ - 회원 3명: 김하준, 이수진, 박민준                                           │
│ - 반려동물 5마리: 바둑이(진돗개), 나비(코리안숏헤어), 초코(푸들),            │
